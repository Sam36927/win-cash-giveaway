    <script>
        // Configuration
        const UPI_ID = 'trtrading1993@oksbi';
        const BUSINESS_NAME = 'Win Cash Giveaway';
        
        // Mock Database (Replace with real database in production)
        let users = JSON.parse(localStorage.getItem('users')) || [];
        let entries = JSON.parse(localStorage.getItem('entries')) || [];
        let currentUser = null;

        // Authentication Handling
        function showForm(formType) {
            document.querySelectorAll('.toggle-btn').forEach(btn => btn.classList.remove('active'));
            document.querySelectorAll('form').forEach(form => form.classList.add('hidden'));
            
            document.querySelector(`#${formType}Form`).classList.remove('hidden');
            event.currentTarget.classList.add('active');
        }

        function handleAuth(e, type) {
            e.preventDefault();
            const elements = e.target.elements;
            
            const userData = {
                name: type === 'signup' ? elements[0].value : null,
                email: type === 'signup' ? elements[1].value : elements[0].value,
                password: type === 'signup' ? elements[2].value : elements[1].value
            };

            // Validation
            if(type === 'signup') {
                if(users.some(u => u.email === userData.email)) {
                    showStatus('Email already registered!', 'error');
                    return;
                }
                users.push({ ...userData, entries: [] });
                localStorage.setItem('users', JSON.stringify(users));
            }
            
            // Login verification
            const user = users.find(u => u.email === userData.email && u.password === userData.password);
            if(!user) {
                showStatus('Invalid credentials!', 'error');
                return;
            }

            currentUser = user;
            showPaymentSection();
        }

        // Payment Handling
        function showPaymentSection() {
            document.getElementById('authSection').classList.add('hidden');
            document.getElementById('paymentSection').classList.remove('hidden');
            
            // Show QR code for PC users
            if(!isMobile()) {
                document.getElementById('qrSection').style.display = 'block';
                generateQRCode();
            }
        }

        function isMobile() {
            return /Android|webOS|iPhone|iPad|iPod|BlackBerry|IEMobile|Opera Mini/i.test(navigator.userAgent);
        }

        function generateQRCode() {
            const upiLink = `upi://pay?pa=${UPI_ID}&pn=${encodeURIComponent(BUSINESS_NAME)}&am=11&tn=GiveawayEntry`;
            const qrUrl = `https://chart.googleapis.com/chart?chs=200x200&cht=qr&chl=${encodeURIComponent(upiLink)}`;
            
            document.getElementById('qrCode').innerHTML = `
                <img src="${qrUrl}" alt="Google Pay QR Code" style="width:100%;height:100%">
            `;
        }

        function initiateGPayPayment() {
            const paymentLink = `upi://pay?pa=${UPI_ID}&pn=${encodeURIComponent(BUSINESS_NAME)}&am=11&tn=GiveawayEntry`;
            
            // Open Google Pay
            window.location.href = paymentLink;
            
            // Start payment verification check
            setTimeout(verifyPayment, 3000);
        }

        function verifyPayment() {
            // In production: Implement actual payment verification
            const isVerified = confirm('Did you complete the payment? (Simulation)');
            
            if(isVerified) {
                handlePaymentSuccess();
            } else {
                showStatus('Payment verification failed. Please try again.', 'error');
            }
        }

        function handlePaymentSuccess() {
            const entryCode = generateUniqueCode();
            
            // Store entry
            entries.push({
                code: entryCode,
                email: currentUser.email,
                timestamp: new Date().toISOString()
            });
            localStorage.setItem('entries', JSON.stringify(entries));
            
            // Show code
            document.getElementById('paymentSection').classList.add('hidden');
            document.getElementById('codeSection').classList.remove('hidden');
            document.getElementById('uniqueCode').textContent = entryCode;
        }

        // Helper Functions
        function generateUniqueCode() {
            return `GW-${Date.now().toString().slice(-8)}-${Math.random().toString(36).substr(2, 3).toUpperCase()}`;
        }

        function showStatus(message, type) {
            const statusEl = document.getElementById('paymentStatus');
            statusEl.className = `status ${type}`;
            statusEl.textContent = message;
            statusEl.style.display = 'block';
        }

        // Admin Function (Run in console)
        function selectWinner() {
            const entries = JSON.parse(localStorage.getItem('entries')) || [];
            if(entries.length === 0) return alert('No entries yet!');
            
            const winner = entries[Math.floor(Math.random() * entries.length)];
            alert(`Winner Code: ${winner.code}\nEmail: ${winner.email}`);
        }
